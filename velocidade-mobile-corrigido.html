// Proxy CORS para a API interna (AxiAgro / Usina Moreno) // Versão Cloudflare Workers — cole este código inteiro no editor do Worker. const API_URL = 'http://52.7.202.88:3333'; // Quem pode usar este proxy. Mantenha a lista curta. const ORIGENS_PERMITIDAS = [ 'http://usinamoreno.s3-website-us-east-1.amazonaws.com', 'http://localhost:3000', 'http://localhost:5500', ]; function origemPermitida(origem) { if (!origem) return true; // sem origem (teste direto no navegador) if (origem === 'null') return true; // arquivo aberto do PC (file://) return ORIGENS_PERMITIDAS.includes(origem); } function cabecalhosCors(origem) { return { 'Access-Control-Allow-Origin': origem || '*', 'Access-Control-Allow-Methods': 'GET,POST,PUT,DELETE,OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type,Authorization', 'Access-Control-Max-Age': '86400', }; } export default { async fetch(request) { const origem = request.headers.get('Origin'); const url = new URL(request.url); const cors = cabecalhosCors(origem); // Rota de teste: abra a URL do Worker + /health no navegador if (url.pathname === '/' || url.pathname === '/health') { return new Response( JSON.stringify({ status: 'ok', destino: API_URL }), { headers: { ...cors, 'Content-Type': 'application/json' } } ); } if (!origemPermitida(origem)) { return new Response( JSON.stringify({ erro: 'Origem não autorizada: ' + origem }), { status: 403, headers: { 'Content-Type': 'application/json' } } ); } // Pré-voo do CORS: o navegador pergunta antes se pode chamar if (request.method === 'OPTIONS') { return new Response(null, { status: 204, headers: cors }); } // Monta o endereço de destino mantendo caminho e parâmetros const destino = API_URL + url.pathname + url.search; try { const cabecalhos = { 'Content-Type': request.headers.get('Content-Type') || 'application/json', }; const auth = request.headers.get('Authorization'); if (auth) cabecalhos['Authorization'] = auth; const resposta = await fetch(destino, { method: request.method, headers: cabecalhos, body: ['GET', 'HEAD'].includes(request.method) ? undefined : await request.text(), }); const texto = await resposta.text(); return new Response(texto, { status: resposta.status, headers: { ...cors, 'Content-Type': resposta.headers.get('Content-Type') || 'application/json', }, }); } catch (e) { return new Response( JSON.stringify({ erro: 'Não consegui falar com a API', detalhe: e.message }), { status: 502, headers: { ...cors, 'Content-Type': 'application/json' } } ); } }, };
